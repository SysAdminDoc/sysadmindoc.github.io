# OWNERSHIP: portfolio static site (sites/portfolio/)
# Route block for portfolio.getparkerai.com. Merged into the live
# /home/deploy/proxy/Caddyfile (mirror: Contabo-VPS-Ops/server-state/Caddyfile).
#
# This is the whole point of moving off GitHub Pages: it serves the static
# portfolio container AND sets the response headers Pages cannot — HSTS,
# X-Frame-Options (clickjacking, since a <meta> CSP cannot deliver
# frame-ancestors), Permissions-Policy, and COOP. The full
# Content-Security-Policy is delivered by the portfolio container's internal
# Caddy response header, while the site's own <meta http-equiv> tag remains the
# source policy used to stamp that header.

portfolio.getparkerai.com {
	encode zstd gzip
	reverse_proxy portfolio-app:80
	header {
		Strict-Transport-Security "max-age=31536000; includeSubDomains"
		X-Content-Type-Options    "nosniff"
		Referrer-Policy           "strict-origin-when-cross-origin"
		X-Frame-Options           "DENY"
		Permissions-Policy        "camera=(), microphone=(), geolocation=(), payment=(), usb=()"
		Cross-Origin-Opener-Policy "same-origin"
		# The internal Caddy delivers the full hash-pinned CSP. This edge-owned
		# endpoint map supplies the absolute Reporting API destination that the
		# CSP report-to directive selects.
		Reporting-Endpoints      "csp-endpoint=\"https://portfolio.getparkerai.com/csp-report\""
	}
	# Rolled at midnight and deleted after 30 days, the period /privacy/ states
	# (src/data/retention.ts). Size-based rolling alone never rolled this
	# low-traffic log, so it had kept every visitor address since the move.
	log {
		output file /var/log/caddy/portfolio.log {
			roll_at 00:00
			roll_keep 30
			roll_keep_for 720h
		}
		format json
	}
}

# OWNERSHIP: portfolio lead notifications (sites/portfolio/, ntfy)
# ntfy for the contact form, served through portfolio-app so ntfy itself stays
# on the portfolio's private compose network. ntfy refuses everything without a
# token, so the edge only terminates TLS. No `encode` here: compressing a
# subscription stream buffers it, and flush_interval -1 keeps it streaming.
notify.getparkerai.com {
	reverse_proxy portfolio-app:80 {
		flush_interval -1
	}
	header {
		Strict-Transport-Security "max-age=31536000; includeSubDomains"
		X-Content-Type-Options    "nosniff"
		Referrer-Policy           "no-referrer"
		X-Frame-Options           "DENY"
		-Server
	}
}
